def merge
  selected = {}
  conflicts = []
  flags.each do |flag|
    chosen, conflict = choose(flag, options(@base, flag), options(@ours, flag), options(@theirs, flag))
    selected[flag] = chosen if chosen
    conflicts << conflict if conflict
  end
  { selected: selected, conflicts: conflicts }
end
